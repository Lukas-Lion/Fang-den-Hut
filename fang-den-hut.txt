      <p>Landest du auf einem normalen Feld bei einer anderen Farbe, werden alle dort stehenden Hütchen bzw. Türme unter dein Hütchen aufgenommen. Das oberste Hütchen kontrolliert den gesamten Turm.</p>
      <h3>Ruhefelder</h3>
      <p>Graue Felder sind sicher: Dort wird nicht gefangen. Maximal drei Hütchen/Türme dürfen gleichzeitig auf einem Ruhefeld stehen.</p>
      <h3>Heimat</h3>
      <p>Ein Hütchen darf nach dem Verlassen nur dann wieder in die eigene Heimat, wenn es Gefangene transportiert. Gefangene Gegner werden dort gewertet; eigene Hütchen im Turm werden befreit. Überschüssige Würfelpunkte werden weitergezogen.</p>
      <h3>Spielende</h3>
      <p><b>Standard:</b> Wenn nur noch eine Farbe aktiv ist, werden auch Gefangene in noch fahrenden Türmen gezählt. Höchste Fangzahl gewinnt. <b>Wer übrig bleibt:</b> Die letzte aktive Farbe gewinnt. <b>Der Turm:</b> Eine Heimkehr ist nicht möglich; die letzte kontrollierende Farbe gewinnt.</p>

  const link=(a,b)=>{ if(!edges[a].includes(b)) edges[a].push(b); if(!edges[b].includes(a)) edges[b].push(a); };
  const angleAt=i=>-Math.PI/2 + (i/ringCount)*Math.PI*2;
  const segment=ringCount/spokes;

  for(let i=0;i<ringCount;i++){
    const a=angleAt(i), safe=(i%segment===Math.floor(segment/2));
    addNode(`r${i}`,cx+Math.cos(a)*ringR,cy+Math.sin(a)*ringR,safe?'safe':'normal',{ring:true,junction:i%segment===0});
  }
  for(let i=0;i<ringCount;i++) link(`r${i}`,`r${(i+1)%ringCount}`);
  addNode('center',cx,cy,'normal',{center:true,junction:true});
  const junctions=[];
  for(let s=0;s<spokes;s++){
    const ringIndex=s*segment; junctions.push(ringIndex);
    const a=angleAt(ringIndex);
    let prev=`r${ringIndex}`;
    for(let j=1;j<=spokeSteps;j++){
      const frac=j/(spokeSteps+1); const rad=ringR*(1-frac);
      const safe = j===3;
      const id=`s${s}_${j}`;
      addNode(id,cx+Math.cos(a)*rad,cy+Math.sin(a)*rad,safe?'safe':'normal',{spoke:s,junction:false});
      link(prev,id); prev=id;
    }
    link(prev,'center');
  }
  const homes=[];
  for(let p=0;p<nPlayers;p++){
    const s=p; const ringIndex=junctions[s]; const a=angleAt(ringIndex);
    homes.push({id:`home-${p}`,playerId:p,entry:`r${ringIndex}`,x:cx+Math.cos(a)*homeR,y:cy+Math.sin(a)*homeR});
  }
  return {nodes,edges,homes,ringCount,spokes,ringR,cx,cy};
}

function enumeratePaths(start, steps, owner, carryingStart){
  const results=[];
  const walk=(node,remaining,prev,nodes,carrying,passedHome)=>{
    if(remaining===0){ results.push({nodes:[...nodes],passedHome,carrying}); return; }
    let neis=[];
    if(node.startsWith('home-')){
      neis=[board.homes[owner].entry];
    }else{
      neis=[...(board.edges[node]||[])];
      if(node===board.homes[owner].entry && carrying && state.mode!=='tower') neis.push(`home-${owner}`);
    }
    for(const next of neis){
      if(next===prev && !node.startsWith('home-')) continue;
      if(next.startsWith('home-') && (!carrying || state.mode==='tower')) continue;
      let nextCarrying=carrying, nextPassed=passedHome;
      if(next.startsWith('home-')){ nextCarrying=false; nextPassed=true; }
      walk(next,remaining-1,node,[...nodes,next],nextCarrying,nextPassed);
    }
  };
  walk(start,steps,null,[start],carryingStart,false);
  const unique=new Map(); for(const p of results){ const end=p.nodes[p.nodes.length-1]; const k=end+'|'+p.passedHome; if(!unique.has(k)) unique.set(k,p); }
  return [...unique.values()];
}
function legalPath(path, owner){
  const end=path.nodes[path.nodes.length-1];
  if(end.startsWith('home-')) return path.passedHome && state.mode!=='tower';
  const node=board.nodes[end]; if(!node) return false;
  if(node.type==='safe' && (state.occupancy[end]?.length||0)>=3) return false;
  return true;
}
function destinationsFrom(kind,id){
  const out=new Map(); if(state.phase!=='move'||!state.die) return out;
  let owner=state.current,start,carrying=false;
  if(kind==='home'){
    if(Number(id)!==owner||state.home[owner]<=0) return out;
    start=`home-${owner}`;
  }else{
    const found=towerById(id); if(!found||topOwner(found.tower)!==owner) return out;
    start=found.nodeId; carrying=found.tower.hats.length>1;
  }
  for(const path of enumeratePaths(start,state.die,owner,carrying)){
    if(!legalPath(path,owner)) continue;
    const end=path.nodes[path.nodes.length-1]; if(!out.has(end)) out.set(end,path);
  }
  return out;
}

function executeMove(move){
  if(state.finished||state.phase!=='move') return;
  pushHistory();
  const owner=state.current, p=player(owner); let movingTower;
  if(move.kind==='home'){
    if(state.home[owner]<=0) return;
    state.home[owner]--; movingTower={id:`t${state.towerSeq++}`,hats:[owner]};
  }else{
    const found=towerById(move.id); if(!found) return; movingTower=deepClone(found.tower);
    state.occupancy[found.nodeId]=state.occupancy[found.nodeId].filter(t=>t.id!==move.id);
  }

  const end=move.path.nodes[move.path.nodes.length-1];
  let deposited=[];
  if(move.path.passedHome){
    deposited=movingTower.hats.filter(h=>h!==owner);
    const own=movingTower.hats.filter(h=>h===owner).length;
    state.bank[owner].push(...deposited);
    const endsAtHome=end.startsWith('home-');
    if(endsAtHome){ state.home[owner]+=own; movingTower=null; }
    else{ state.home[owner]+=Math.max(0,own-1); movingTower={id:movingTower.id,hats:[owner]}; }
    if(deposited.length) addLog(`<b>${esc(p.name)}</b> bringt ${deposited.length} gefangene${deposited.length===1?'s Hütchen':' Hütchen'} nach Hause.`);
    if(own>1) addLog(`${own-1} eigene Hütchen von <b>${esc(p.name)}</b> werden befreit.`);
  }

  let captured=0;
  if(movingTower && !end.startsWith('home-')){
    const node=board.nodes[end], occupants=ensureNode(end);
    if(node.type!=='safe' && occupants.length){
      const foreign=occupants.some(t=>topOwner(t)!==owner);
      if(foreign){
        const caught=[]; for(const t of occupants) caught.push(...t.hats);
        captured=countEnemyHats(caught,owner);
        movingTower.hats=[...caught,...movingTower.hats];
        state.occupancy[end]=[];
        if(captured) addLog(`<b>${esc(p.name)}</b> fängt ${captured} gegnerische${captured===1?'s Hütchen':' Hütchen'}!`);
      }
    }
    ensureNode(end).push(movingTower);
  }

  state.lastMove={owner,dest:end,captured,deposited:deposited.length,die:state.die};
  beep(captured?780:510,captured?.12:.06,captured?.045:.028);
  selected=null; legalDestinations=new Map();
  const rolledSix=state.die===6; state.die=null;
  const ended=checkGameEnd();
  if(!ended){
    if(rolledSix){ state.phase='roll'; addLog(`<b>${esc(p.name)}</b> darf wegen der 6 noch einmal würfeln.`); }
    else nextPlayer();
  }
  save(); renderAll();
  if(!state.finished && currentPlayer().type==='bot') scheduleBot();
}

}

  let n=0; for(const {tower} of allTowers()) if(topOwner(tower)===owner) n+=countEnemyHats(tower.hats,owner); return n;

function checkGameEnd(){
  const active=activeControllers();
  if(state.turn < state.players.length) return false;
  if(active.length>1) return false;
  state.finished=true; state.phase='finished';
  const ranking=state.players.map((p,i)=>({id:i,name:p.name,color:p.color,score:scoreFor(i),active:active.includes(i)}));
  if(state.mode==='standard') ranking.sort((a,b)=>b.score-a.score||Number(b.active)-Number(a.active));
  else ranking.sort((a,b)=>Number(b.active)-Number(a.active)||b.score-a.score);

function botChooseMove(moves){
  if(state.finished||currentPlayer().type!=='bot'||state.phase!=='move') return;
  let best=-1e9, options=[];
  for(const m of moves){
    let score=Math.random()*1.2; const owner=state.current;
    let movingHats=[owner]; if(m.kind==='tower'){ const f=towerById(m.id); if(f) movingHats=f.tower.hats; }
    if(m.path.passedHome){ score+=countEnemyHats(movingHats,owner)*12 + 5; }
    if(!m.dest.startsWith('home-')){
      const node=board.nodes[m.dest]; const occ=state.occupancy[m.dest]||[];
      if(node.type==='safe') score+=2.2;
      else if(occ.some(t=>topOwner(t)!==owner)){
        let cap=0; for(const t of occ) cap+=countEnemyHats(t.hats,owner)+1; score+=cap*10;
      }
      score += graphSafetyScore(m.dest,owner)*.35;
    }
    if(score>best+.01){best=score;options=[m];} else if(Math.abs(score-best)<.01) options.push(m);
  }
  const pick=options[Math.floor(Math.random()*options.length)]||moves[0]; executeMove(pick);
}
function graphSafetyScore(nodeId,owner){
  if(board.nodes[nodeId]?.type==='safe') return 5;
  let danger=0;
  const q=[[nodeId,0,null]],seen=new Set([nodeId]);
  while(q.length){ const [n,d]=q.shift(); if(d>=6) continue; for(const nx of board.edges[n]||[]){ if(seen.has(nx)) continue; seen.add(nx); q.push([nx,d+1,n]); const occ=state.occupancy[nx]||[]; if(occ.some(t=>topOwner(t)!==owner)) danger++; } }
  return -danger;
}

  if(state.finished){$('turnSub').textContent='Die Partie ist beendet.';$('phaseBadge').textContent='Ende';$('hintText').innerHTML='Die Partie ist beendet. Du kannst das Brett noch ansehen oder eine neue Partie starten.';return;}
  const bot=p.type==='bot';
  if(state.phase==='roll'){
    $('turnSub').textContent=bot?'Computer denkt …':'Würfle, um deinen Zug zu beginnen.'; $('phaseBadge').textContent='Würfeln'; $('hintText').innerHTML=bot?`<strong>${esc(p.name)}</strong> übernimmt diesen Zug automatisch.`:`Klicke auf <strong>Würfeln</strong>. Bei einer 6 darfst du danach noch einmal würfeln.`;
  }else{
    const moves=allLegalMoves(); $('turnSub').textContent=`Gewürfelt: ${state.die}`; $('phaseBadge').textContent=`🎲 ${state.die}`;
    if(bot) $('hintText').innerHTML=`<strong>${esc(p.name)}</strong> wählt gerade einen Zug …`;
    else if(!moves.length) $('hintText').innerHTML=`Mit der <strong>${state.die}</strong> gibt es keinen legalen Zug. Du musst aussetzen.`;
    else if(!selected) $('hintText').innerHTML=`Wähle ein <strong>Hütchen oder einen Turm</strong>. Danach werden alle erreichbaren Zielfelder grün markiert.`;
    else $('hintText').innerHTML=`Wähle jetzt eines der <strong>grün markierten Zielfelder</strong>.`;
  }
}
